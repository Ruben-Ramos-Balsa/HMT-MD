\section{Spin current and algebraic Cartan--Holst response}
\label{vii:sec:corriente-respuesta-cartan}

The realization of enriched transport in
Section~\ref{vii:sec:realizacion-cartan} provides the co-tetrad and
connection; the dimensional sections and the incidence parameter in
Sections~\ref{v:ant:sec:barbero} and~\ref{v:ant:sec:gravity} fix their
couplings. Their composition determines a first-order variational
problem. The matter current occupies a precise position within it:
it is the derivative of the action with respect to the connection and
determines the torsional response with the same normalization.

\subsection{Geometric domain and current normalization}
\label{vii:subsec:dominio-corriente-cartan}

We work on a smooth oriented four-dimensional manifold \(M\),
with internal Lorentzian bundle \(E\), metric
\(\eta=\operatorname{diag}(-1,1,1,1)\), and nondegenerate co-tetrad
\(e\in\Omega^1(M;E)\). In a local frame, write
\(e^I\), with \(I=0,1,2,3\). The metric connection
\(\omega^{IJ}=-\omega^{JI}\) acts through \(D_\omega\).
When the fields \(\Psi\) are spinorial, a spin structure and the
corresponding matter representation are retained. Define
\begin{equation}
 \Sigma^{IJ}=e^I\wedge e^J,\qquad
 F^{IJ}=\mathrm d\omega^{IJ}+\omega^I{}_{K}\wedge\omega^{KJ},
 \qquad T^I=D_\omega e^I.
 \label{vii:eq:campos-cartan}
\end{equation}
The operator \(\star\) acts on internal bivector indices,
\begin{equation}
 (\star X)^{IJ}=\tfrac12\epsilon^{IJ}{}_{KL}X^{KL},
 \qquad \star^2=-I,\qquad
 \mathsf P_\gamma=\star+\gamma^{-1}I.
 \label{vii:eq:operador-holst}
\end{equation}
This duality is thus distinguished from the exterior duality of forms
on \(M\). Internal indices are raised and contracted with \(\eta\).

Fix a chart with \(G_{\rm int}>0\), \(c_{\rm int}>0\),
\(\gamma=\gamma_{\rm HMT}\in\mathbb R\setminus\{0\}\), and
\(\Lambda_{\rm int}\) constant on it. Abbreviate
\(\kappa=8\pi G_{\rm int}/c_{\rm int}^4\) and \(c=c_{\rm int}\).
These sections remain fixed during variation. The action is
\begin{equation}
 \begin{split}
 S[e,\omega,\Psi]={}&
 \frac1{2\kappa c}\int_M\Sigma_{IJ}\wedge
                         (\mathsf P_\gamma F)^{IJ}
 -\frac{\Lambda_{\rm int}}{\kappa c}
                         \int_M\operatorname{vol}_e
 +S_{\rm m}[e,\omega,\Psi].
 \end{split}
 \label{vii:eq:accion-cartan-holst}
\end{equation}
The volume and orientation convention is the one giving
\(\Sigma_{IJ}\wedge(\star F)^{IJ}=R\operatorname{vol}_e\)
in the Levi--Civita sector. The preceding dimensional type makes each
summand an action section.

Connection variations have compact support in the interior, or satisfy
\(\delta\omega|_{\partial M}=0\). A boundary completion whose
variational problem cancels the same boundary term may also be used.
At fixed co-tetrad and matter fields, the spin current is the
three-form with values in dual internal bivectors determined by
\begin{equation}
 \delta_\omega S_{\rm m}
 =-\tfrac12\int_M\delta\omega^{IJ}\wedge\sigma_{IJ}.
 \label{vii:eq:corriente-variacional}
\end{equation}
Pairing with an arbitrary compactly supported variation uniquely
determines \(\sigma\). This definition retains the action unit of
the current and the normalization factor entering its geometric response.

\subsection{Connection variation and Holst inversion}
\label{vii:subsec:variacion-holst}

\begin{theorem}[Variational Cartan--Holst equation]
\label{vii:thm:ecuacion-cartan-holst}
On the preceding domain, stationarity with respect to the connection
is equivalent to
\begin{equation}
 D_\omega(\mathsf P_\gamma\Sigma)^{IJ}
       =\kappa c\,\sigma^{IJ}.
 \label{vii:eq:ecuacion-cartan-holst}
\end{equation}
\end{theorem}
\begin{proof}
The curvature variation is
\(\delta F=D_\omega\delta\omega\). Internal duality is parallel
and self-adjoint for bivector contraction; since \(\gamma\) is
constant, \(D_\omega\mathsf P_\gamma=\mathsf P_\gamma D_\omega\).
Setting \(B_{IJ}=(\mathsf P_\gamma\Sigma)_{IJ}\) gives
\[
 B_{IJ}\wedge D_\omega\delta\omega^{IJ}
 =\mathrm d(B_{IJ}\wedge\delta\omega^{IJ})
   -D_\omega B_{IJ}\wedge\delta\omega^{IJ}.
\]
The last form changes sign when its factors of degrees three and one
are interchanged. The boundary conditions eliminate the integral of
the exact term. The cosmological term is independent of \(\omega\).
Therefore,
\[
 \delta_\omega S=
 \frac1{2\kappa c}\int_M\delta\omega^{IJ}\wedge
       \bigl[D_\omega B_{IJ}-\kappa c\,\sigma_{IJ}\bigr].
\]
The arbitrariness of the variation proves the equation, including its
sign and coefficient.
\end{proof}

\begin{lemma}[Real inverse of the Holst operator]
\label{vii:lem:inversa-holst}
For \(\gamma\in\mathbb R\setminus\{0\}\),
\begin{equation}
 \mathsf P_\gamma^{-1}
 =\frac{\gamma^2}{1+\gamma^2}
                      (-\star+\gamma^{-1}I).
 \label{vii:eq:inversa-holst}
\end{equation}
\end{lemma}
\begin{proof}
Both factors are polynomials in \(\star\) and commute. Their
product before normalization is
\[
 (\star+\gamma^{-1}I)(-\star+\gamma^{-1}I)
 =-\star^2+\gamma^{-2}I=(1+\gamma^{-2})I.
\]
The denominator \(1+\gamma^2\) is positive on the fixed real
domain. Multiplication by \(\gamma^2/(1+\gamma^2)\) gives the identity.
\end{proof}

The current thus determines the bivector-valued three-form
\begin{equation}
 Y^{IJ}:=\kappa c\,
   \frac{\gamma^2}{1+\gamma^2}
   (-\star+\gamma^{-1}I)\sigma^{IJ},
 \qquad D_\omega\Sigma^{IJ}=Y^{IJ}.
 \label{vii:eq:corriente-dualizada}
\end{equation}

\subsection{Pointwise reconstruction of torsion and contorsion}
\label{vii:subsec:reconstruccion-torsion}

The Leibniz rule gives
\begin{equation}
 D_\omega\Sigma^{IJ}
 =T^I\wedge e^J-e^I\wedge T^J
 =:\mathsf L_e(T)^{IJ}.
 \label{vii:eq:mapa-torsion}
\end{equation}
At each point, \(\mathsf L_e\) acts from
\(\Lambda^2T^*M\otimes E\) to
\(\Lambda^3T^*M\otimes\Lambda^2E\). Both spaces have dimension
twenty-four. Let \(E_I\) be the frame dual to the co-tetrad, so
that \(\iota_{E_I}e^J=\delta_I^J\).

\begin{theorem}[Algebraic inversion of the torsional equation]
\label{vii:thm:inversion-torsion}
For a nondegenerate co-tetrad, \(\mathsf L_e\) is an isomorphism.
Its inverse on the form \(Y\) is written
\begin{equation}
 B^I=\sum_J\iota_{E_J}Y^{IJ},\qquad
 b=\tfrac14\sum_I\iota_{E_I}B^I,\qquad
 T^I=B^I-e^I\wedge b.
 \label{vii:eq:torsion-explicita}
\end{equation}
Equation~\eqref{vii:eq:ecuacion-cartan-holst} therefore determines
torsion pointwise from \(e\) and the current appearing in it.
\end{theorem}
\begin{proof}
First let \(Y=\mathsf L_e(T)\) and set
\(t=\sum_J\iota_{E_J}T^J\). Contracting
\eqref{vii:eq:mapa-torsion}, using the fact that \(T^I\) has
degree two and that the co-tetrad has four elements, gives
\[
 \sum_J\iota_{E_J}(T^I\wedge e^J)=2T^I,
 \qquad
 \sum_J\iota_{E_J}(e^I\wedge T^J)=T^I-e^I\wedge t.
\]
Thus \(B^I=T^I+e^I\wedge t\). A second contraction yields
\[
 \sum_I\iota_{E_I}B^I=t+3t=4t.
\]
This recovers \(t=b\) and then \(T^I=B^I-e^I\wedge b\).
In particular, the kernel of \(\mathsf L_e\) is zero. Equality
of dimensions proves surjectivity and makes the formula valid for
every form \(Y\) in the codomain.
\end{proof}

The connection decomposes uniquely as
\begin{equation}
 \omega^{IJ}=\mathring\omega^{IJ}(e)+K^{IJ},\qquad
 K^{IJ}=-K^{JI},\qquad T^I=K^I{}_{J}\wedge e^J,
 \label{vii:eq:contorsion}
\end{equation}
where \(\mathring\omega(e)\) is the Levi--Civita connection
and \(K\) is contorsion. To make the inverse explicit, write
\(K_{IJ}=K_{IJK}e^K\) and
\(T_I=\tfrac12T_{IJK}e^J\wedge e^K\). Then
\begin{equation}
 T_{IJK}=K_{IKJ}-K_{IJK},\qquad
 K_{IJK}=\tfrac12(T_{KIJ}-T_{IJK}-T_{JKI}).
 \label{vii:eq:contorsion-componentes}
\end{equation}
The second equality follows by cyclically combining the first and
using \(K_{IJK}=-K_{JIK}\); substitution recovers the components
of \(T\). Torsion and contorsion are consequently two equivalent
presentations of the algebraic response at fixed co-tetrad.

\begin{corollary}[Zero-current sector and minimal linear response]
\label{vii:cor:desacoplamiento-torsional}
If \(\sigma=0\), then \(T=0\) and
\(\omega=\mathring\omega(e)\). For a matter coupling whose
current \(\sigma[e,\Psi]\) is independent of \(K\),
equations~\eqref{vii:eq:corriente-dualizada},
\eqref{vii:eq:torsion-explicita}, and
\eqref{vii:eq:contorsion-componentes} give a unique response
\(K_*(e,\Psi)\), linear in \(\sigma\).
\end{corollary}
\begin{proof}
The three reconstruction maps are linear at fixed \(e\),
\(\gamma\), and \(\kappa c\). Zero current produces
\(Y=0\), then \(T=0\), and finally \(K=0\).
\end{proof}

In the minimal action, the connection equation contains this pointwise
response. Its evolution follows by evolving \(e\) and \(\Psi\):
\(K_*=K_*(e,\Psi)\) changes with them. If a matter action retains
additional algebraic dependence on \(K\), the Cartan equation
retains its variational form and must be solved with that current;
the preceding linear response belongs to the specified minimal domain.

\subsection{Elimination of contorsion and quadratic contribution}
\label{vii:subsec:respuesta-cuadratica}

Now consider the minimal coupling affine in \(K\), expressed by
\begin{equation}
 S_{\rm m}[e,\mathring\omega+K,\Psi]
 =S_{\rm m}[e,\mathring\omega,\Psi]
   -\tfrac12\int_M K^{IJ}\wedge\sigma_{IJ}[e,\Psi].
 \label{vii:eq:materia-afin-contorsion}
\end{equation}
This condition identifies the matter action to which quadratic
elimination applies; the current remains the one defined by
\eqref{vii:eq:corriente-variacional}.

The curvature expansion is
\begin{equation}
 F(\mathring\omega+K)
 =F(\mathring\omega)+D_{\mathring\omega}K+K\wedge K,
 \qquad (K\wedge K)^{IJ}=K^I{}_{L}\wedge K^{LJ}.
 \label{vii:eq:curvatura-contorsion}
\end{equation}
Since \(D_{\mathring\omega}\Sigma=0\), the linear term is
the exterior derivative of \(\Sigma_{IJ}\wedge(\mathsf P_\gamma K)^{IJ}\).
The corresponding boundary contribution remains explicit:
\begin{equation}
 S_\partial[e,K]=\frac1{2\kappa c}
     \int_{\partial M}\Sigma_{IJ}\wedge(\mathsf P_\gamma K)^{IJ}.
 \label{vii:eq:borde-contorsion}
\end{equation}
On an interior domain, or with the compatible boundary completion,
the action density depending on contorsion is
\begin{equation}
 \mathcal Q_{e,\gamma}(K)-\tfrac12K^{IJ}\wedge\sigma_{IJ},
 \qquad
 \mathcal Q_{e,\gamma}(K)
  =\frac1{2\kappa c}\Sigma_{IJ}\wedge
                         \mathsf P_\gamma(K\wedge K)^{IJ}.
 \label{vii:eq:forma-cuadratica-contorsion}
\end{equation}
The density \(\mathcal Q_{e,\gamma}\) is a differential
four-form, homogeneous of degree two in \(K\).

\begin{theorem}[Effective response quadratic in the current]
\label{vii:thm:respuesta-cuadratica}
Under \eqref{vii:eq:materia-afin-contorsion}, substitution of the
algebraic solution \(K_*\) produces the effective spin density
\begin{equation}
 \mathcal L_{\rm spin}^{\rm eff}
 =-\tfrac14 K_*^{IJ}\wedge\sigma_{IJ}
 =-\mathcal Q_{e,\gamma}(K_*).
 \label{vii:eq:accion-efectiva-spin}
\end{equation}
At fixed co-tetrad and couplings, it is homogeneous of degree two in
the current. The effective action also retains
\(S_\partial[e,K_*]\), or its declared boundary completion.
\end{theorem}
\begin{proof}
By Corollary~\ref{vii:cor:desacoplamiento-torsional},
\(K_*\) is linear in \(\sigma\). The connection equation is
precisely the stationarity of
\eqref{vii:eq:forma-cuadratica-contorsion}. Since this is algebraic,
the radial variation \(\delta K=K_*\) applies pointwise;
it may also be multiplied by a compactly supported function.
Euler's identity for a quadratic form gives
\[
 2\mathcal Q_{e,\gamma}(K_*)
             -\tfrac12K_*^{IJ}\wedge\sigma_{IJ}=0.
\]
Substitution into the action density proves
\eqref{vii:eq:accion-efectiva-spin}. If
\(\sigma\mapsto\lambda\sigma\), then
\(K_*\mapsto\lambda K_*\) and the effective density is
multiplied by \(\lambda^2\). The expansion
\eqref{vii:eq:curvatura-contorsion} separately identifies the
boundary term accompanying this elimination.
\end{proof}

The specific current, its Lorentzian contraction, duality, and the
matter-action conventions determine the sign and coefficient of this
contribution. The quadratic identity provides the composition rule
that evaluates them once these data are specified.

\subsection{Effective metric source and covariant balance}
\label{vii:subsec:fuente-metrica-efectiva}

The normalization of the metric source is fixed in the same action
line. After eliminating \(K\), define
\begin{align}
 \delta_g S_{\rm m}[e,\mathring\omega,\Psi]
 &=-\tfrac12\int_M\operatorname{vol}_e\,
                   \mathcal T^{S,\rm LC}_{\mu\nu}\,\delta g^{\mu\nu},
 \\
 \delta_g S_{\rm spin}^{\rm eff}
 &=-\tfrac12\int_M\operatorname{vol}_e\,
                   \mathcal U^{S,\rm spin}_{\mu\nu}\,\delta g^{\mu\nu}.
 \label{vii:eq:fuentes-normalizadas-accion}
\end{align}
Here interior variations are considered, with boundary terms treated
according to the fixed variational problem. For affine coupling, the
second source is the metric variation of
\eqref{vii:eq:accion-efectiva-spin}. The variation includes the
dependence of the matter current on \(e\) and \(\Psi\).

\begin{proposition}[Metric equation and conservation of the total source]
\label{vii:prop:fuente-metrica-balance}
With the preceding conventions, the metric equations are
\begin{equation}
 G_{\mu\nu}[g]+\Lambda_{\rm int}g_{\mu\nu}
 =\kappa c\bigl(\mathcal T^{S,\rm LC}_{\mu\nu}
                    +\mathcal U^{S,\rm spin}_{\mu\nu}\bigr)
 =\kappa\bigl(T^{\rm phys,LC}_{\mu\nu}
                    +U^{\rm phys,spin}_{\mu\nu}\bigr),
 \label{vii:eq:ecuacion-metrica-efectiva}
\end{equation}
where \(T^{\rm phys,LC}=c\mathcal T^{S,\rm LC}\) and
\(U^{\rm phys,spin}=c\mathcal U^{S,\rm spin}\) when the
co-tetrad takes values in the length line. Every solution satisfies
\begin{equation}
 \mathring\nabla^\mu
   (T^{\rm phys,LC}_{\mu\nu}+U^{\rm phys,spin}_{\mu\nu})=0.
 \label{vii:eq:balance-metrico-total}
\end{equation}
\end{proposition}
\begin{proof}
Eliminating \(K\) preserves the equations for the remaining
variables: in the chain rule, the term multiplying
\(\delta K_*\) vanishes by the connection equation. In the
Levi--Civita sector, the first Bianchi identity makes the term
\(\Sigma_{IJ}\wedge F^{IJ}\) vanish, and the gravitational
part is written
\((2\kappa c)^{-1}\int_M(R-2\Lambda_{\rm int})\operatorname{vol}_e\).
Its interior variation is
\[
 \frac1{2\kappa c}\int_M\operatorname{vol}_e\,
        (G_{\mu\nu}+\Lambda_{\rm int}g_{\mu\nu})\delta g^{\mu\nu}.
\]
Adding \eqref{vii:eq:fuentes-normalizadas-accion} gives
\eqref{vii:eq:ecuacion-metrica-efectiva}. The contracted Bianchi
identity for \(\mathring\nabla\), metric compatibility, and
constancy of \(\kappa\) and \(\Lambda_{\rm int}\) in the
chart yield \eqref{vii:eq:balance-metrico-total}.
\end{proof}

Before eliminating the connection, the geometric identities retain
the form
\begin{equation}
 D_\omega T^I=F^I{}_{J}\wedge e^J,\qquad D_\omega F^{IJ}=0.
 \label{vii:eq:bianchi-cartan}
\end{equation}
The first is \(D_\omega^2e=F\wedge e\); the second follows
by expanding \(D_\omega(\mathrm d\omega+\omega\wedge\omega)\)
and cancelling terms. In a covariant matter action, total metric
balance agrees with the Noether identity on the matter equations.
Geometry and matter thus retain a joint source; the division between
its Levi--Civita contribution and its spin contribution is determined
by the specialized action.
