\chapter{Cohomological image and composition toward Hodge}

\section{Source object, finite observations and publication}

Let $X$ be a smooth complex projective variety and $p\ge0$. The object produced by
the common construction and received here is the complete cohomological structure
$\mathfrak G_{\rm coh}(X,p)$, which preserves
based charts, oriented crossings, Mayer--Vietoris cones, carry,
boundary, route, multisection, survival and terminal. At depth $N$ we
write
\begin{equation}\label{eq:pdf2-hodge-sistemas}
 \mathscr S_N(X,p):=\operatorname{Surv}_{J_N}(X,p),
 \qquad
 \rho_{N+1,N}:\mathscr S_{N+1}(X,p)\to\mathscr S_N(X,p),
\end{equation}
and the modules of observations
\begin{equation}\label{eq:pdf2-hodge-observaciones}
 e_N:\mathscr S_N(X,p)\to\mathscr H_N^p(X),
 \qquad
 q_{N+1,N}:\mathscr H_{N+1}^p(X)\to\mathscr H_N^p(X),
\end{equation}
with the square
\begin{equation}\label{eq:pdf2-hodge-cuadrado-restriccion}
 e_N\rho_{N+1,N}=q_{N+1,N}e_{N+1}.
\end{equation}
For
\[
 \alpha\in\operatorname{Hdg}^p(X)_{\mathbb Q}
 :=H^{2p}(X,\mathbb Q)\cap H^{p,p}(X),
\]
let $q_N\alpha$ be its family of finite observations. The fiber to be
prolonged is
\begin{equation}\label{eq:pdf2-hodge-fibra-alpha}
 \mathscr F_N(\alpha):=
 \{s\in\mathscr S_N(X,p):e_N(s)=q_N\alpha\}.
\end{equation}

The producer and the publication are typed by
\begin{equation}\label{eq:pdf2-hodge-realizador}
 R_{\rm Hdg}:X_\chi(X,p)\to
 K_0\!\left(\operatorname{Perf}^{\ge p}(X)\right)_{\mathbb Q},
\end{equation}
\begin{equation}\label{eq:pdf2-hodge-publicacion}
 \operatorname{cl}_{X,p}\circ\operatorname{ch}^{\rm loc}_p
 \circ R_{\rm Hdg}
 =\operatorname{ev}^{\rm Hdg}_{\infty,X,p}.
\end{equation}
This is the reader action identity already constructed within the complete object
\eqref{eq:pdf2-g-coh-limite-lector-dato}; it does not introduce the
preimage of a Hodge class from the codomain. The preceding exact owner constructs that
existence, and this chapter unfolds the composition. The reader
must remain in the order
\[
 \mathfrak G_{\rm coh}(X,p)
 \longrightarrow X_\chi(X,p)
 \xrightarrow{R_{\rm Hdg}}
 K_0\!\left(\operatorname{Perf}^{\ge p}(X)\right)_{\mathbb Q}
 \xrightarrow{\operatorname{ch}^{\rm loc}_p}
 \operatorname{CH}^p(X)_{\mathbb Q}
 \xrightarrow{\operatorname{cl}_{X,p}}
 \operatorname{Hdg}^p(X)_{\mathbb Q}.
\]
In particular, the final class does not retrospectively select a cell or a cycle.

\section{Relative module and triangular incidence table}

Having fixed $N$, let
\begin{equation}\label{eq:pdf2-hodge-kernels}
 \mathscr K^{\rm S}_{N+1}:=\ker\rho_{N+1,N},
 \qquad
 \mathscr K^{\rm H}_{N+1}:=\ker q_{N+1,N},
 \qquad
 e_{N+1}^{\rm rel}:=e_{N+1}|_{\mathscr K^{\rm S}_{N+1}}.
\end{equation}
The new charts $\Delta J_{N+1}$ produce
$\mathcal C_\nu=\operatorname{Cone}(\kappa_{a,b})$ and the module
\begin{equation}\label{eq:pdf2-hodge-a-rel}
 \mathscr A_{N+1}^{\rm rel}:=
 H^0\!\left(
  \mathsf W_{\rm rel}\otimes^{\mathbf L}_{\mathbb Q[\Delta J_{N+1}]}
  \Gamma_{\rm rel}
  \oplus\operatorname{Cone}(I-U_{\Gamma_9})_{\rm rel}
 \right).
\end{equation}
Generator maps
\begin{equation}\label{eq:pdf2-hodge-mapas-relativos}
 a_{N+1}:\mathscr A_{N+1}^{\rm rel}\to\mathscr K^{\rm S}_{N+1},
 \qquad
 b_{N+1}:\mathscr A_{N+1}^{\rm rel}\to\mathscr K^{\rm H}_{N+1}
\end{equation}
satisfy
\begin{equation}\label{eq:pdf2-hodge-factorizacion-relativa}
 e_{N+1}^{\rm rel}a_{N+1}=b_{N+1}.
\end{equation}

\subsection{Generator-by-generator presentation of the new chart}

So that \eqref{eq:pdf2-hodge-factorizacion-relativa} is not a black box,
the two finite sets involved are written explicitly. Let
\(\mathcal G_{N+1}^{\rm coh}\) be the set of all new labels
\[
 \nu=(Z,W,a,b,\operatorname{or},\operatorname{Carr},
       \operatorname{Mem},\partial,\gamma,\tau^{\rm term})
\]
that appear in \(\Sigma_{\rm coh}^{\rm mult}\), including one terminal
label for each component of
\(\operatorname{Cone}(I-U_{\Gamma_9})_{\rm rel}\). Let
\(\mathcal R_{N+1}^{\rm coh}\) be the set of three classes of relations:
Mayer--Vietoris at crossings \(Z\cap W\), change of orientation/carry, and
terminal compatibility of one complete turn. Form
\begin{equation}\label{eq:pdf2-hodge-presentacion-finita}
\begin{aligned}
 F_{N+1}^{\rm coh}&:=\mathbb Q^{(\mathcal G_{N+1}^{\rm coh})},&
 D_{N+1}&:\mathbb Q^{(\mathcal R_{N+1}^{\rm coh})}
             \longrightarrow F_{N+1}^{\rm coh},\\
 \mathscr A_{N+1}^{\rm rel}
   &\cong F_{N+1}^{\rm coh}/\operatorname{im}D_{N+1}.&&
\end{aligned}
\end{equation}
The action of the realizer on the base is not left implicit:
\begin{equation}\label{eq:pdf2-hodge-rhdg-generadores}
\begin{aligned}
 R_{{\rm Hdg},N+1}(e_\nu)
   &:=[\operatorname{Cone}(\kappa_{a,b}^{Z,W})],
       &&\nu\ \text{cellular},\\
 R_{{\rm Hdg},N+1}(e_{\nu_{\rm term}})
   &:=[\operatorname{Cone}(I-U_{\Gamma_9})_{\nu_{\rm term}}],
       &&\nu_{\rm term}\ \text{terminal}.
\end{aligned}
\end{equation}
The additivity of cones in the Mayer--Vietoris triangle, the identity
\(\kappa_{b,a}\tau=-P\kappa_{a,b}\), the carry cocycle, and the identity of
terminal return give, respectively for the three classes of relations,
\begin{equation}\label{eq:pdf2-hodge-descenso-relaciones}
 R_{{\rm Hdg},N+1}D_{N+1}=0
 \quad\text{in }K_0(\operatorname{Perf}^{\ge p}(X))_{\mathbb Q}.
\end{equation}
Therefore, \eqref{eq:pdf2-hodge-rhdg-generadores} descends to the quotient of
\eqref{eq:pdf2-hodge-presentacion-finita}; surjectivity of the cycle class has not yet
been assumed.

Once the ordered basis
\(\{\omega_\lambda\}_{\lambda\in\Lambda_{N+1}}\) of new observations
published by \(\mathscr K^{\rm H}_{N+1}\) is fixed, the computable matrix of the reader is
\begin{equation}\label{eq:pdf2-hodge-matriz-lector}
 C_{N+1}(\lambda,\nu):=
 \left\langle\omega_\lambda^\vee,\,
 \operatorname{cl}_{X,p}\operatorname{ch}^{\rm loc}_p
 R_{{\rm Hdg},N+1}(e_\nu)\right\rangle .
\end{equation}
From \eqref{eq:pdf2-hodge-descenso-relaciones} is obtained
\begin{equation}\label{eq:pdf2-hodge-cd-cero}
 C_{N+1}D_{N+1}=0.
\end{equation}
Thus, the entries that follow are not scalar ranks of the cones: they are their
complete cohomological coordinates in the basis \(\omega_\lambda\).

\begin{proposition}[Finite exhaustiveness of the cohomological owner]
\label{prop:pdf2-hodge-exhaustividad-owner}
The coordinate $\Sigma_{\rm coh}^{\rm mult}$ of
$\mathfrak G_{\rm coh}$ induces an exhaustive decomposition of
$\mathscr K^{\rm H}_{N+1}$ into new-cylinder atoms and a terminal
difference. Each cylinder atom is the principal incidence of a cone
$\operatorname{Cone}(\kappa_{a,b})$ and occurs with coefficient one; the
remaining summand is exactly
$H^0(\operatorname{Cone}(I-U_{\Gamma_9})_{\rm rel})$.
\end{proposition}

\begin{proof}
Exhaustiveness and the unitary table are constructed before this view: the
finite extensions come from
\eqref{eq:amp-hodge-surv-n}--\eqref{eq:amp-hodge-ext}, and their compatibility is
proved in \cref{prop:amp-hodge-extension-finita}. The coinductive step and
evaluation are derived in
\eqref{eq:amp-hodge-xi-limite}--\eqref{eq:amp-hodge-evaluaciones-xi};
\eqref{eq:pdf2-g-coh-tabla-dato} records that construction in the common
codomain. In passing from $J_N$ to $J_{N+1}$, an observation either
restricts to an earlier one, has support in a cell of
$\Delta J_{N+1}$, or measures the difference of a complete turn. The
three cases are disjoint by support, route and terminal. The last two
enumerate exactly \(\mathcal G_{N+1}^{\rm coh}\) in
\eqref{eq:pdf2-hodge-presentacion-finita}. The cone orientation fixes the coefficient
of its principal incidence at one, and
\eqref{eq:pdf2-hodge-descenso-relaciones} identifies the related copies.
Therefore each element of the basis
\(\{\omega_\lambda\}_{\lambda\in\Lambda_{N+1}}\) has a unique maximal
label and all other entries of its column are strictly earlier.
No class \(\alpha\) has been enumerated, nor has an algebraic output
been chosen retrospectively.
\end{proof}

Each new-remark atom preserves the complete label
\[
 \mu=(Z,W,a,b,\operatorname{or},\operatorname{Carr},
       \operatorname{Mem},\partial,\gamma,\tau^{\rm term}).
\]
Order these labels by depth, support, sheet, route, carry,
memory, and terminal. Let $\bar c_\mu$ be the atoms resulting from
$\mathscr K^{\rm H}_{N+1}$ and $c_\mu$ the cellular normal-form representatives in
$\mathscr A_{N+1}^{\rm rel}$. The localized character of a
cone publishes its new incidence with coefficient one; its proper faces
have an earlier label. Therefore there exists a finite strictly
triangular table
\begin{equation}\label{eq:pdf2-hodge-tabla-triangular}
 b_{N+1}(c_\mu)
 =\bar c_\mu+\sum_{\nu<\mu}B_{\nu\mu}\bar c_\nu,
 \qquad B_{\nu\mu}\in\mathbb Q.
\end{equation}
The terminal generator appears as a label in its own right and is not reduced to its
visible phase. The relation
$\kappa_{b,a}\tau=-P\kappa_{a,b}$ fixes the signs of
$B_{\nu\mu}$, and the carry cocycle fixes their composition.

\begin{proposition}[Section relative constructed before the class]
\label{prop:pdf2-hodge-seccion-relativa}
Recurrence
\begin{equation}\label{eq:pdf2-hodge-sigma-recursiva}
 \begin{aligned}
  \sigma_{N+1}^{\rm rel}(\bar c_{\mu_0})&:=c_{\mu_0},\\
  \sigma_{N+1}^{\rm rel}(\bar c_\mu)&:=
  c_\mu-\sum_{\nu<\mu}B_{\nu\mu}
             \sigma_{N+1}^{\rm rel}(\bar c_\nu)
 \end{aligned}
\end{equation}
and the $\mathbb Q$-linear extension define
\begin{equation}\label{eq:pdf2-hodge-sigma-right-inverse}
 \sigma_{N+1}^{\rm rel}:\mathscr K^{\rm H}_{N+1}\longrightarrow
 \mathscr A_{N+1}^{\rm rel},
 \qquad b_{N+1}\sigma_{N+1}^{\rm rel}=I.
\end{equation}
Consequently,
\begin{equation}\label{eq:pdf2-hodge-right-inverse}
 s_{N+1}^{\rm rel}:=a_{N+1}\sigma_{N+1}^{\rm rel}:
 \mathscr K^{\rm H}_{N+1}\longrightarrow\mathscr K^{\rm S}_{N+1},
 \qquad e_{N+1}^{\rm rel}s_{N+1}^{\rm rel}=I.
\end{equation}
None of these maps depends on $\alpha$.
\end{proposition}

\begin{proof}
The formula \eqref{eq:pdf2-hodge-tabla-triangular} shows that the
matrix of $b_{N+1}$ on the subspace generated by the $c_\mu$ is
unitriangular. For the first label one has
$b\sigma(\bar c_{\mu_0})=\bar c_{\mu_0}$. If the identity holds for all
labels preceding $\mu$, then
\[
 b\sigma(\bar c_\mu)
 =\bar c_\mu+\sum_{\nu<\mu}B_{\nu\mu}\bar c_\nu
  -\sum_{\nu<\mu}B_{\nu\mu}\bar c_\nu
 =\bar c_\mu.
\]
The induction proves \eqref{eq:pdf2-hodge-sigma-right-inverse}; it uses neither the
injectivity of the presentation module nor an auxiliary vanishing result. Applying
\eqref{eq:pdf2-hodge-factorizacion-relativa} gives
\[
 e_{N+1}^{\rm rel}s_{N+1}^{\rm rel}
 =e_{N+1}^{\rm rel}a_{N+1}\sigma_{N+1}^{\rm rel}
 =b_{N+1}\sigma_{N+1}^{\rm rel}=I.
\]
The table is formed from all the cells of
$\Sigma_{\rm coh}^{\rm mult}$ and the terminal; it therefore does not select a
particular atlas after the observation is known.
\end{proof}

\section{Naturalness and the universal extension operator}

An admissible refinement preserves the complete labels and replaces a
cell with its Mayer--Vietoris expression with terms of strictly
lower order. If $r_{\mathscr A}$, $r_{\rm H}$, and $r_{\rm S}$ are the
induced maps, then
\begin{equation}\label{eq:pdf2-hodge-entrelazadores-refinamiento}
 b' r_{\mathscr A}=r_{\rm H}b,
 \qquad
 a' r_{\mathscr A}=r_{\rm S}a,
 \qquad
 e'^{\rm rel}r_{\rm S}=r_{\rm H}e^{\rm rel}.
\end{equation}
The global normal form preserves the first pivot of each label. The
uniqueness of the recurrence \eqref{eq:pdf2-hodge-sigma-recursiva} implies
\begin{equation}\label{eq:pdf2-hodge-naturalidad}
 r_{\mathscr A}\sigma_{N+1}^{\rm rel}
 =\sigma_{N+1}^{\prime\,\rm rel}r_{\rm H},
 \qquad
 r_{\rm S}s_{N+1}^{\rm rel}
 =s_{N+1}^{\prime\,\rm rel}r_{\rm H}.
\end{equation}
Thus orientation, carry, memory, boundary, route, and terminal are preserved.

The unitary degeneration that forms part of
\eqref{eq:pdf2-g-coh-datos-operativos} provides, before any class is fixed,
\begin{equation}\label{eq:pdf2-hodge-extension-cero}
 \iota_{N+1,N}:\mathscr S_N(X,p)\longrightarrow\mathscr S_{N+1}(X,p),
 \qquad \rho_{N+1,N}\iota_{N+1,N}=I.
\end{equation}
Define the fiber product of compatible data.
\begin{equation}\label{eq:pdf2-hodge-pullback-extension}
 \mathscr P_{N+1,N}:=
 \{(s,h)\in\mathscr S_N(X,p)\times\mathscr H_{N+1}^p(X):
             e_N(s)=q_{N+1,N}(h)\}.
\end{equation}
For $(s,h)\in\mathscr P_{N+1,N}$, the defect
\begin{equation}\label{eq:pdf2-hodge-defecto-universal}
 \delta_{N+1}(s,h):=
 h-e_{N+1}\iota_{N+1,N}(s)
 \in\mathscr K^{\rm H}_{N+1}
\end{equation}
it is corrected by the material formulas
\begin{equation}\label{eq:pdf2-hodge-eta-extension}
 \begin{aligned}
  \eta_{N+1}(s,h)&:=
     s_{N+1}^{\rm rel}\bigl(\delta_{N+1}(s,h)\bigr),\\
  \operatorname{Ext}_{N+1,N}(s,h)&:=
     \iota_{N+1,N}(s)+\eta_{N+1}(s,h).
 \end{aligned}
\end{equation}

\begin{theorem}[Finite universal and natural extension]
\label{thm:pdf2-hodge-extension-universal}
For every compatible datum $(s,h)$,
\begin{equation}\label{eq:pdf2-hodge-dos-identidades-extension}
 \rho_{N+1,N}\operatorname{Ext}_{N+1,N}(s,h)=s,
 \qquad
 e_{N+1}\operatorname{Ext}_{N+1,N}(s,h)=h.
\end{equation}
The operator commutes with every admissible refinement and is defined without choosing an algebraic cycle.
\end{theorem}

\begin{proof}
The square \eqref{eq:pdf2-hodge-cuadrado-restriccion} and the fibered-product
condition give
\[
 q_{N+1,N}\delta_{N+1}(s,h)
 =q_{N+1,N}h-e_Ns=0,
\]
so that \eqref{eq:pdf2-hodge-eta-extension} is typed. Since
$\eta_{N+1}$ belongs to $\ker\rho_{N+1,N}$, the first identity of
\eqref{eq:pdf2-hodge-dos-identidades-extension} follows from
\eqref{eq:pdf2-hodge-extension-cero}. For the second one uses
\eqref{eq:pdf2-hodge-right-inverse}:
\[
 e_{N+1}\operatorname{Ext}_{N+1,N}(s,h)
 =e_{N+1}\iota_{N+1,N}(s)
  +e_{N+1}^{\rm rel}s_{N+1}^{\rm rel}\delta_{N+1}(s,h)=h.
\]
Naturality is the composition of
\eqref{eq:pdf2-hodge-naturalidad} with the naturality of the unitary
degeneration. The input $h$ is a cohomological remark; at no step
is it presupposed to be the class of a cycle.
\end{proof}

\section{Initial fiber, Mittag--Leffler and affirmative publication}

At depth zero, the based section constructed in
\eqref{eq:amp-hodge-seccion-base} and published in
\eqref{eq:pdf2-g-coh-seccion-dato} assigns to each atom $c_\mu$ of
$\mathscr H_0^p(X)$ a seed state $\widehat c_\mu$. This assignment
belongs to the structural completeness of the cohomological image and is fixed
before choosing $\alpha$; it is not an algebraic cycle representing the
final class. Therefore
\begin{equation}\label{eq:pdf2-hodge-seccion-base}
 s_0^{\rm base}\!\left(\sum_\mu t_\mu c_\mu\right)
 :=\sum_\mu t_\mu\widehat c_\mu,
 \qquad e_0s_0^{\rm base}=I.
\end{equation}
For $\alpha\in\operatorname{Hdg}^p(X)_{\mathbb Q}$, define
recursively
\begin{equation}\label{eq:pdf2-hodge-recursion-alpha}
 s_0:=s_0^{\rm base}(q_0\alpha),
 \qquad
 s_{N+1}:=\operatorname{Ext}_{N+1,N}(s_N,q_{N+1}\alpha).
\end{equation}

\begin{theorem}[Coinductive Completeness and Hodge Realization]
\label{thm:pdf2-hodge-completitud-final}
The family \eqref{eq:pdf2-hodge-recursion-alpha} verifica
\[
 s_N\in\mathscr F_N(\alpha),
 \qquad \rho_{N+1,N}s_{N+1}=s_N.
\]
Consequently, the fibres form a nonempty Mittag--Leffler system. More
strongly, the recursion exhibits a concrete compatible section at every
level; no \(\varprojlim^1\) of relative kernels is invoked. Moreover,
\begin{equation}\label{eq:pdf2-hodge-xi-alpha}
 \xi_\alpha:=(s_N)_{N\ge0}
 \in\varprojlim_N\mathscr F_N(\alpha)\subset X_\chi(X,p).
\end{equation}
The first identity of \eqref{eq:pdf2-g-coh-limite-lector-dato} materially identifies
the compatible family with a point of $X_\chi(X,p)$. The
cylindrical projections, including the terminal difference, separate classes by
the same datum; therefore
\begin{equation}\label{eq:pdf2-hodge-evaluacion-alpha}
 \operatorname{ev}^{\rm Hdg}_{\infty,X,p}(\xi_\alpha)=\alpha.
\end{equation}
Finally,
\begin{equation}\label{eq:pdf2-hodge-beta}
 \beta_\alpha:=\operatorname{ch}^{\rm loc}_p
 (R_{\rm Hdg}(\xi_\alpha))\in\operatorname{CH}^p(X)_{\mathbb Q}
\end{equation}
satisfies
\begin{equation}\label{eq:pdf2-hodge-clase-final}
 \operatorname{cl}_{X,p}(\beta_\alpha)=\alpha.
\end{equation}
\end{theorem}

\begin{proof}
The identity $e_0s_0^{\rm base}=I$ initiates the induction. If $e_Ns_N=q_N\alpha$, the pair $(s_N,q_{N+1}\alpha)$ belongs to \eqref{eq:pdf2-hodge-pullback-extension}; theorem~\ref{thm:pdf2-hodge-extension-universal} simultaneously gives the required restriction and evaluation. This proves the nonemptiness and surjectivity of the fiber transitions and hence the Mittag--Leffler condition. The explicit compatible family defines $\xi_\alpha$ through \eqref{eq:pdf2-g-coh-limite-lector-dato}. Its evaluation and $\alpha$ coincide in every finite projection and at the terminal; separation, also included in that datum, identifies them. Applying \eqref{eq:pdf2-hodge-publicacion} produces \eqref{eq:pdf2-hodge-clase-final}.
\end{proof}

\section{Non-governing relative control and falsifiers}

The presentation cokernel
\begin{equation}\label{eq:pdf2-hodge-o-rel}
 \mathscr O_{N+1}^{\rm rel}:=\operatorname{coker}
 (d_{\rm coend}:\mathscr R_{N+1}^{\rm rel}\to
                  \mathscr P_{N+1}^{\rm rel})
\end{equation}
and the comparator
\begin{equation}\label{eq:pdf2-hodge-chi-rel}
 \chi_{N+1}^{\rm rel}:\mathscr O_{N+1}^{\rm rel}\to
 \mathscr K^{\rm H}_{N+1}
\end{equation}
are redundancy controls for the presentation. The preceding proof
does not assume $\ker\chi_{N+1}^{\rm rel}=0$: a redundant relation may live
in that kernel without preventing the extension. The table
\eqref{eq:pdf2-hodge-tabla-triangular} and
$b\sigma=I$ does force
$\operatorname{coker}\chi_{N+1}^{\rm rel}=0$, as a subsequent control.

\begin{falsifier}[Local falsifiers of the construction]
The proof is refuted if a remark label is missing from $\Sigma_{\rm coh}^{\rm mult}$, if a diagonal coefficient of \eqref{eq:pdf2-hodge-tabla-triangular} is not one, if a refinement changes the first global pivot, if the terminal generator is removed, or if $e^{\rm rel}a=b$ fails. None of these controls uses a final algebraic class as an input.
\end{falsifier}
