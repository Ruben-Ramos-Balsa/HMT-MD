  CATALOGO_IDIOMA=ES

  cp -R "05_REPRODUCCION/OBRAS/VIa/$CATALOGO_IDIOMA" "$CATALOGO_TRABAJO/$CATALOGO_IDIOMA"

  python3 -I -S "$CATALOGO_TRABAJO/$CATALOGO_IDIOMA/technical/preparar_catalogo_masas.py" --rev2-root "05_REPRODUCCION/DATOS/MASAS/REV2" --integral-root "05_REPRODUCCION/DATOS/MASAS/INTEGRAL"

  $CatalogoIdioma = 'ES'
