This appendix gathers the proofs that can be verified without reconstructing the entire narration. The hypotheses appear in each statement. The classical results used to interpret an operator that has already been constructed are distinguished from the internal identities.

\section{Positivity of the vacuum operator}

\begin{lemma}
If \(T=T^*\) and \(\Spec(T)\subset(0,1)\), then \(I-T^n>0\) for every \(n\ge1\), \(I-T^{120}\) is invertible and
\[
\mathcal V=(I-T^{90})(I-T^{120})^{-1}>0.
\]
\end{lemma}

\begin{proof}
For \(t\in(0,1)\), \(1-t^n>0\). By functional calculus, \(I-T^n\) is positive. Since the spectrum of \(I-T^{120}\) is separated from zero in finite dimension, it has a positive inverse. Both factors are functions of \(T\), commute, and their product is positive.
\end{proof}

\begin{lemma}[Monotonicity of the completion]
The function
\[
r(q)=\frac{1-q^{90}}{1-q^{120}}
\]
is strictly decreasing in \((0,1)\).
\end{lemma}

\begin{proof}
With \(s=q^{30}\),
\[
r(s)=\frac{1+s+s^2}{1+s+s^2+s^3}.
\]
A derivation yields
\[
\frac{dr}{ds}=
-\frac{s^2(3+2s+s^2)}{(1+s+s^2+s^3)^2}<0.
\]
As \(ds/dq=30q^{29}>0\), also \(dr/dq<0\).
\end{proof}

\section{Traces that recover the constituent characters}

From \(\log\mathcal V=(\log r_+)P_++(\log r_-)P_-\) and \(\tau(P_\pm)=1/2\),

\[
2\tau(\log\mathcal V)=\log r_++\log r_-=\log(r_+r_-).
\]

Moreover, \(RP_+=P_+\), \(RP_-=-P_-\); hence

\[
-2\tau(R\log\mathcal V)
=-\log r_++\log r_-
=\log(r_-/r_+).
\]

Exponentiating yields \(\widehat\mu,\widehat\varepsilon,\widehat Z,\widehat c\). There is no approximation in this step.

\section{UHF Tower Compatibility}

Be \(\mathcal A_m=M_2\otimes M_{9^m}\) and the inclusion \(\iota_m(x)=x\otimes I_9\). The expectativa

\[
E_m=\mathrm{id}_{M_2}\otimes\mathrm{id}_{M_{9^m}}\otimes\tau_9
\]

satisfies \(E_m\iota_m=\mathrm{id}\). Therefore,

\[
E_m(\mathcal V\otimes I_{9^{m+1}})
=\mathcal V\otimes I_{9^m}.
\]

The same equality holds for every continuous function of \(\mathcal V\), in particular \(\log\mathcal V\). The normalized trace is compatible, and the characters do not change under refinement.

\section{Coefficients of the analytic criticality profile}

Let

\[
b_m=\frac{2(3m-1)}{\sqrt{899}}.
\]

The expansion \(1-\cos(b_mx)=b_m^2x^2/2-b_m^4x^4/24+O(x^6)\) gives

\[
u_0(x)=\frac1{24}\sum_{m=1}^{12}b_m^2x^2
-\frac1{288}\sum_{m=1}^{12}b_m^4x^4+O(x^6).
\]

Consequently,

\[
\sum b_m^2=\frac4{899}\sum(3m-1)^2=24,
\]

whence the quadratic coefficient is one. The integer calculation of the fourth moment reduces the coefficient to

\[
-\frac{715769}{2424603}.
\]

This fraction provides an algebraic control more informative than its
decimal value.

\section{Closed trigonometric sum}

The identity

\[
\sum_{m=1}^{M}\cos(a+(m-1)d)
=\frac{\sin(Md/2)}{\sin(d/2)}
\cos\!\left(a+\frac{M-1}{2}d\right)
\]

with \(M=12\), \(a=4y\), \(d=6y\) produces

\[
\sum_{m=1}^{12}\cos(2(3m-1)y)
=\frac{\sin36y}{\sin3y}\cos37y.
\]

Dividing by \(12\) proves the first form of \cref{mab:rm:eq:chaos-closed}; \(2\sin36y\cos37y=\sin73y-\sin y\) proves the second.

\section{Schwarziano negativo}

For \(x\in(0,1]\), each \(b_mx\in(0,\pi)\). Therefore

\[
u_0'(x)=\frac1{12}\sum b_m\sin(b_mx)>0,
\]

\[
u_0'''(x)=-\frac1{12}\sum b_m^3\sin(b_mx)<0.
\]

Since \(f_\lambda'=-\lambda u_0'\), the factors \(-\lambda\) cancel in the Schwarzian quotients, and

\[
S(f_\lambda)=\frac{u_0'''}{u_0'}
-\frac32(u_0''/u_0')^2<0.
\]

Parity extends the conclusion to \([-1,0)\). The origin is critical quadratic by \(u_0(x)=x^2+O(x^4)\).

\section{Levy integral}

The dominated convergence allows

\begin{align*}
-\int_0^1\log x\,d\mu_G(x)
&=\frac1{\log2}\int_0^1\frac{-\log x}{1+x}\,dx\\
&=\frac1{\log2}\sum_{n=0}^\infty(-1)^n
\int_0^1x^n(-\log x)\,dx\\
&=\frac1{\log2}\sum_{n=1}^\infty\frac{(-1)^{n-1}}{n^2}\\
&=\frac{\eta(2)}{\log2}
=\frac{\pi^2}{12\log2}.
\end{align*}

\section{Reconstruction of the areal and modular sectors}

Let

\[
N=N_{90}+N_{120},\qquad D=90N_{90}+120N_{120}.
\]

The system matrix is

\[
M=\begin{pmatrix}1&1\\90&120\end{pmatrix},
\qquad \det M=30.
\]

Therefore,

\[
M^{-1}=\frac1{30}
\begin{pmatrix}120&-1\\-90&1\end{pmatrix},
\]

and

\[
N_{90}=4N-D/30,\qquad N_{120}=D/30-3N.
\]

The determinant \(30\) explains why the fundamental modular step reappears in the infinite classification.

\section{Purification through a thermofield double state}

For a qutrit with weight \(w>0\),

\[
Z(w)=1+\ee^{-w}+\ee^{-2w},
\qquad p_n=\frac{\ee^{-nw}}{Z(w)}.
\]

Define

\[
|\mathrm{TFD}(w)\rangle=\sum_{n=0}^2\sqrt{p_n}
|n\rangle_A|n\rangle_{\bar A}.
\]

Then

\[
\Tr_{\bar A}|\mathrm{TFD}\rangle\langle\mathrm{TFD}|
=\sum_np_n|n\rangle\langle n|=\rho_A.
\]

The equality of the entropies of the two subsystems is immediate from the Schmidt decomposition. The product of eighteen copies of each channel purifies the total boundary state.

\section{1st areal modular law}

For a differentiable family \(\rho(t)\), \(\Tr\rho(t)=1\), with faithful reference state \(\rho_0\) and \(K_0=-\log\rho_0\),

\[
\left.\frac{d}{dt}S(\rho(t))\right|_{t=0}
=\Tr(\dot\rho_0K_0),
\]

porque \(\Tr\dot\rho_0=0\). Si

\[
K_0-cI=\beta_{90}\mathcal A_{90}
+\beta_{120}\mathcal A_{120},
\]

the scalar term vanishes, and

\[
\delta S=\beta_{90}\delta\langle\mathcal A_{90}\rangle
+\beta_{120}\delta\langle\mathcal A_{120}\rangle.
\]

As \(\beta_m=m\Dmod/(\gammaH a_0)\), the ratio is exactly \(120/90=4/3\).

\section{Classification of the modular realization in the infinite limit}

Suppose an infinite succession of qutrit factors with faithful states, whose logarithms of eigenvalue ratios belong to

\[
90\Dmod\Z\quad\text{or}\quad120\Dmod\Z,
\]

and suppose that both types appear with positive density. The additive group generated is

\[
90\Dmod\Z+120\Dmod\Z=30\Dmod\Z.
\]

The asymptotic ratio is then

\[
\{0\}\cup\exp(-30\Dmod\Z)
=\{0\}\cup\lambda_\partial^\Z,
\qquad \lambda_\partial=\ee^{-30\Dmod}.
\]

The Araki--Woods theorem for ITPFI factors classifies the GNS closure as the hyperfinite factor \(III_{\lambda_\partial}\). The application is conditional on the hypotheses of faithfulness, compatibility, and persistence; the weights entering the classification are internal.

\begin{remark}
If only the channel \(90\) persists, the generating group changes to \(90\Dmod\Z\); if only \(120\) persists, it changes to \(120\Dmod\Z\). If the local states are not of the declared compatible product type, the criterion must be recomputed. These cases are refutation criteria, not stylistic notes.
\end{remark}

\[

\frac{m_W}{m_Z}=\frac{\chi(94,-1)}{\chi(95,-1)}

Since \(D_A=\ee^{-2A}=\det T\),

\[
(m_W/m_Z)^2=D_A,\qquad

The proof belongs to the angular chart. Its complete interpretation of
