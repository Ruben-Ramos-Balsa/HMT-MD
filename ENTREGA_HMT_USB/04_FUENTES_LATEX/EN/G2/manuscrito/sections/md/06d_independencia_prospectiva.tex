\section{Criterion of causal independence with respect to external references: directed acyclic graph and prior fixation of the output}
\label{sec:definiciones-independencia-prospectiva}

Let \(\mathsf P\) be the internal primitives, \(\mathsf R\) the declared rules,
readers, gauges, and normalizations, \(G\) a generator,
\(o=G(\mathsf P,\mathsf R)\) its output, and \(y\) an external reference.

\begin{definition}[Prospectively independent interface]
\label{def:interfaz-prospectivamente-independiente}
The interface is prospectively independent of \(y\) if:
\begin{enumerate}
 \item \(y\notin\operatorname{Anc}(o)\);
 \item \(\mathsf P,\mathsf R,G\) and the negative controls are fixed before
 \(y\) is opened;
 \item the execution has no direct or indirect access to \(y\);
 \item \(o\) and its trace are sealed before computing a comparison
 \(\operatorname{cmp}(o,y)\).
\end{enumerate}
\end{definition}

\begin{definition}[Selection independence]
\label{def:independencia-seleccion}
The blindness of each execution is not enough if, after producing candidates
\(G_j(\mathsf P,\mathsf R_j)\), one publishes
\[
 j_\ast(y)=\arg\min_j
 \operatorname{cmp}\!\left(G_j(\mathsf P,\mathsf R_j),y\right).
\]
Selection independence prohibits this operation from determining the published output.
\end{definition}

\begin{definition}[Multivalued prediction]
\label{def:prediccion-multivaluada}
If the non-metrological physical descriptors and the internal constraints
admit several multisections, the prospective output is the complete set
\[
 \widehat{\mathfrak R}(d)
 =\{\mathcal M:\mathcal M\text{ satisfies all the constraints
 preregistered for }d\}.
\]
Nonuniqueness is an output of the procedure. Subsequently choosing an element
by proximity to a magnitude is prohibited.
\end{definition}

\section{No-Interference Theorem}
\label{sec:teorema-no-interferencia}

\begin{definition}[Directed Acyclic Faithful and Complete Causal Graph]
\label{def:dag-fiel-completo}
A directed acyclic graph \(D\) is faithful if every node represents the same function and the same inputs as the cited construction. It is complete for an output if every causally relevant dependency appears as an edge or a declared root. Faithfulness and completeness are semantic hypotheses; the acyclicity of the dataset does not prove them.
\end{definition}

\begin{theorem}[Prospective independence conditional on the causal graph]
\label{thm:no-interferencia-prospectiva}
Let a deterministic system of structural equations be represented by a
finite directed acyclic graph \(D\) that is faithful and complete for \(o\). If
\(y\notin\operatorname{Anc}_D(o)\), all ancestral roots of \(o\)
remain frozen and the execution has no hidden access to \(y\), then
for any \(y_0,y_1\),
\[
 o_{\operatorname{do}(y=y_0)}
 =o_{\operatorname{do}(y=y_1)}.
\]
\end{theorem}

\begin{proof}
The predecessors of \(o\) are topologically ordered. Their roots are internal
and remain fixed. By induction on the order, each node receives the same
arguments under both interventions and, by determinism, returns the same
value. Equality propagates to \(o\). The cryptographic digest does not create
the independence; it merely makes a subsequent alteration detectable.
\end{proof}

\begin{corollary}[Permutation of the external comparison]
\label{cor:permutacion-comparacion-externa}
If names, rows, or external values are permuted after sealing, the
internal output does not change, although its score may change:
\[
 o_{\operatorname{do}(y=\sigma y_0)}
 =o_{\operatorname{do}(y=y_0)},\qquad
 \operatorname{cmp}(o,\sigma y_0)
 \ne\operatorname{cmp}(o,y_0).
\]
\end{corollary}

\begin{proposition}[Calibrated branch separation]
\label{prop:separacion-ramas-calibradas}
If an external reference intervened in a historical selection, the
dependence does not contaminate internal objects that do not descend from
that selection. The branch can be isolated and subjected to a new
prospective protocol without modifying the kernel.
\end{proposition}

\begin{proof}
To be a predecessor is a local property of the graph. The branches
\[
 Z_0\longrightarrow p=5\longrightarrow\text{electronic scale},
\qquad
 \{\text{masses}\}\longrightarrow\{\text{calibrated signatures}\}
 \longrightarrow\text{table}
\]
remain outside the protected subgraph. The towers, the formal character, the
atlas, and the center \((5,5)\) receive no edges from those references.
\end{proof}

\begin{algorithm}[Future assignment without magnitudes]
\label{alg:asignacion-futura-sin-magnitudes}
The physical interface is executed in four irreversible phases:
\begin{enumerate}
 \item \emph{Freezing}: authority, atlas, involutions,
restrictions, code, priority, and negative controls are sealed.
 \item \emph{Permitted input}: only sector, generation, charge,
 color, conjugation, spin/statistics, composition, and metrological type are

 \item \emph{Output}: all compatible multisections are enumerated; for
 each extension of each multisection, the observable route, record,
 signature, \(\chi_{\rm form}\), specialization, and observable are computed. In composite
 objects, the extensions and records are spliced first. The complete
 multiset is sealed.
 \item \emph{Opening}: a distinct custodian opens the magnitudes and computes
residuals, uncertainties, and covariances without modifying any candidate.
\end{enumerate}
\end{algorithm}

The enriched multisection, rather than a point extension selected
retrospectively, is the output of the physical matching. The subsequent
evaluation may be performed fiber by fiber. The preceding algorithm is a
falsifiable specification; it is not presented as an already executed multiset
prediction.

\label{crit:fallo-protocolo-ciego}
The protocol fails if the code can read a magnitude, if a calibrated
historical signature reappears as an input, if a candidate is discarded after
opening the reference, if positive mass is confused with exactly zero mass,
or if the sealed multiset changes when the external catalog is permuted.
\end{criterion}

\begin{theorem}[Positive reconstructive completeness]
\label{thm:completitud-reconstructiva-positiva}
Let
\[
 \mathcal N=\langle90,120\rangle_{>0}
\]
and let \((n_j)_{j\ge1}\) be its increasing enumeration. Given
\(0<q_+<q_-<1\), define
\[
 a_j=q_-^{n_j}-q_+^{n_j}>0.
\]
Then \(a_j\to0\) and \(\sum_ja_j<\infty\). Having fixed a deterministic rule
for resolving half-integer ties, for every \(r\in\RR\) the recurrence
\[
 r_0=r,\qquad
 \nu_j=\operatorname{nint}\!\left(\frac{r_{j-1}}{a_j}\right),\qquad
 r_j=r_{j-1}-\nu_ja_j
\]
satisfies
\[
 |r_j|\leq\frac{a_j}{2}\longrightarrow0,\qquad
 r=\sum_{j\ge1}\nu_ja_j,\qquad
 \sum_{j\ge1}|\nu_ja_j|<\infty.
\]
Therefore every \(R>0\), with \(r=\log R\), is represented exactly in the
completion by
\(\exp(\sum_j\nu_ja_j)\).
\end{theorem}

\begin{proof}
The indices of \(\mathcal N\) are \(90,120\) and all multiples of \(30\)
from \(180\) onward. Consequently,
\[
 \sum_{n\in\mathcal N}q_-^n
 =q_-^{90}+q_-^{120}
 +\frac{q_-^{180}}{1-q_-^{30}}<\infty.
\]
As \(0<a_j<q_-^{n_j}\), the sequence is sumable and converge to zero. The
definition of integer mas proximo implica
\(|r_j|\le a_j/2\). By telescopado,
\(\sum_{i=1}^{N}\nu_ia_i=r-r_N\to r\). For \(j\ge2\),
\[
 |\nu_ja_j|=|r_{j-1}-r_j|
 \le\frac{a_{j-1}+a_j}{2},
\]
and the summability of \((a_j)\) proves absolute convergence.
\end{proof}

The theorem establishes completeness of the dictionary on \(\RR_{>0}\). If the
coefficients \(\nu_j\) are obtained from the residue of a known mass, their use
is \textsc{calibrated reconstruction}; it does not provide prospective
selection. Nor does it represent exactly zero mass. The photon, gluon, or a
prospective spin-two mode require exact symmetry protection in the physical
representation.
